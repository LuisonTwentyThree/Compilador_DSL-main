/*

 * =========================================================================

import java.io.IOException;
import java.util.List;

public class CompiladorConASM {
    
    // Atributos existentes
    private MotorLexico motorLexico;
    private MotorSintactico motorSintactico;
    private MotorSemantico motorSemantico;
    private GeneradorCGI generadorCGI;
    
    // NUEVO: Agregar generadores de salida
    private GeneradorEnsamblador generadorASM;
    private IntegradorSalidaASM integrador;
    
    /**
     * Constructor modificado
     */
    public CompiladorConASM() {
        this.motorLexico = new MotorLexico();
        this.motorSintactico = new MotorSintactico();
        this.motorSemantico = new MotorSemantico();
        this.generadorCGI = new GeneradorCGI();
        
        // NUEVO: Inicializar generadores de salida
        this.generadorASM = new GeneradorEnsamblador();
        this.integrador = new IntegradorSalidaASM("salida_dsl");
    }
    
    /**
     * Método principal compilar (modificado)
     */
    public void compilar(String rutaArchivo) throws IOException {
        System.out.println("=== INICIANDO COMPILACIÓN ===");
        
        // 1. ANÁLISIS LÉXICO
        System.out.println("\n[1] Análisis Léxico...");
        List<Token> tokens = motorLexico.analizar(rutaArchivo);
        System.out.println("✓ " + tokens.size() + " tokens generados");
        
        // 2. ANÁLISIS SINTÁCTICO
        System.out.println("\n[2] Análisis Sintáctico...");
        NodoAST ast = motorSintactico.analizar(tokens);
        System.out.println("✓ AST generado");
        
        // 3. ANÁLISIS SEMÁNTICO
        System.out.println("\n[3] Análisis Semántico...");
        motorSemantico.analizar(ast);
        System.out.println("✓ Análisis semántico completado");
        
        // 4. GENERACIÓN DE CÓDIGO INTERMEDIO (3 Direcciones)
        System.out.println("\n[4] Generación de Código Intermedio...");
        List<Cuadruplo> cuadruplos = generadorCGI.generar(ast);
        System.out.println("✓ " + cuadruplos.size() + " cuádruplos generados");
        
        // NUEVO: 5. GENERACIÓN DE ENSAMBLADOR
        System.out.println("\n[5] Generación de Código Ensamblador...");
        generadorASM.procesarCuadruplos(cuadruplos);
        System.out.println("✓ Código ensamblador generado");
        
        // NUEVO: 6. INTEGRACIÓN Y GENERACIÓN DE ARCHIVOS
        System.out.println("\n[6] Integrando y generando archivos de salida...");
        integrador.procesarTodos();
        integrador.generarArchivoASM();
        integrador.generarArchivoVisualizacion();
        System.out.println("✓ Archivos generados exitosamente");
        
        System.out.println("\n=== COMPILACIÓN COMPLETADA ===");
    }
    
    /**
     * Imprime el código generado
     */
    public void imprimirCodigos() {
        System.out.println("\n" + "═".repeat(70));
        System.out.println("CÓDIGO INTERMEDIO (CUÁDRUPLOS)");
        System.out.println("═".repeat(70));
        generadorCGI.imprimirCodigo();
        
        System.out.println("\n" + "═".repeat(70));
        System.out.println("CÓDIGO ENSAMBLADOR");
        System.out.println("═".repeat(70));
        System.out.println(generadorASM.obtenerCodigoEnsamblador());
    }
}

/*
 * PASO 6: Ejemplo de Salida Completa
 * ===================================
 * 
 * ENTRADA DSL:
 * ────────────────────────────────────────────────────
 * estructura arbol miArbol;
 * AGREGARNODO 50 "raiz" EN arbol;
 * AGREGARNODO 30 "izq" EN arbol;
 * AGREGARNODO 70 "der" EN arbol;
 * 
 * CUÁDRUPLOS GENERADOS:
 * ────────────────────────────────────────────────────
 * 0: AGREGARNODO    50                  raiz             -> arbol
 * 1: DIBUJAR_ARBOL  (vacío)             (vacío)          -> arbol
 * 2: AGREGARNODO    30                  izq              -> arbol
 * 3: DIBUJAR_ARBOL  (vacío)             (vacío)          -> arbol
 * 4: AGREGARNODO    70                  der              -> arbol
 * 5: DIBUJAR_ARBOL  (vacío)             (vacío)          -> arbol
 * 
 * CÓDIGO ENSAMBLADOR (extracto):
 * ────────────────────────────────────────────────────
 * main:
 *     push rbp
 *     mov rsp, rbp
 *     
 *     ; AGREGARNODO: clave=50, valor=raiz EN arbol
 *     mov rax, [50]
 *     mov rbx, [raiz]
 *     call agregar_nodo_arbol
 *     call dibujar_arbol
 *     
 *     ... (más instrucciones)
 *     
 *     mov rax, 60
 *     mov rdi, 0
 *     syscall
 * 
 * VISUALIZACIÓN (archivo .txt):
 * ────────────────────────────────────────────────────
 * ╔════════════════════════════════════════╗
 * ║     ÁRBOL BINARIO DE BÚSQUEDA          ║
 * ╚════════════════════════════════════════╝
 * 
 *     ├── [50 : raiz] h=2
 *     │   ├── [30 : izq] h=1
 *     │   └── [70 : der] h=1
 * 
 * ┌─ Estadísticas del Árbol ─────────────┐
 * │ Altura: 2
 * │ Balance Raíz: 0
 * │ Total de Nodos: 3
 * │ Nodos Hoja: 2
 * └──────────────────────────────────────┘
 */
